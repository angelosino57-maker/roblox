# Plan 2: Wesen – Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Eier ausbrüten, eigene Wesen mit Genen, Brüten mit Vererbung und Mutation, bis zu 3 Begleiter mit Boni (Glück, Grabkraft, Tragkraft) und ein Garten mit Münz-Einkommen (auch offline).

**Architecture:** Reine Logik (`Genetics`, `Economy`, erweitertes `Stats`) in `ReplicatedStorage.Shared`, voll getestet. Server-Services `HatchService`, `BreedService`, `CreatureService` (Begleiter + Garten + Einkommen + Modelle) verwalten den Zustand im Profil. Der Client bekommt alles über den bestehenden State-Snapshot und bedient ein einfaches Panel „Wesen“.

**Tech Stack:** Luau, Studio-MCP, Sync über `tools/manifest.py` + `ServerStorage.DevTools.Sync`, Prüfungen über `ServerStorage.DevBridge` (siehe Plan 1, Abschnitt „Abweichungen“).

**Spec:** `docs/superpowers/specs/2026-09-29-dig-and-breed-design.md`, Abschnitt 4. Teil 2 von 4.

**Umfang dieses Plans:** Voller Code für reine Logik und Tests. Services und UI sind über exakte Schnittstellen, Regeln und Prüfungen spezifiziert (der Code entsteht beim Umsetzen), um doppelte Arbeit zu sparen. Optik bleibt schlicht (Nutzerwunsch: erst Mechanik, Design später).

## Global Constraints

- Wesen: `{id: string, species: string, rarity: string, genes: {luck, dig, carry, income: int 1–100}, mutation: string?, generation: int}`
- Effektivwert = `gen × Seltenheits-Multiplikator × Mutations-Multiplikator`.
- Seltenheit: Gewöhnlich 1 / Ungewöhnlich 1,2 / Selten 1,5 / Episch 2 / Legendär 3 / Mythisch 5.
- Mutationen: Gold ×1,5 | Diamant ×2 | Frost ×2 | Lava ×2,5 | Regenbogen ×3 | Galaxy ×4 (Würfelgewichte 40/20/20/12/6/2).
- Begleiter max. 3. `luck = min(5, 1 + Σ eff. luck / 300)`; Grabzeit × `1 / (1 + Σ eff. dig / 400)`; Rucksack + `floor(Σ eff. carry / 20)`.
- Garten: 10 Plätze, `eff. income × 0,1` Münzen/s; offline max. 8 h zu 50 %. Lager max. 100 Wesen.
- Schlüpfen: Gene je Zone Erde 5–25, Stein 15–40, Kristall 30–60; Mutation 1 %; Brutzeit 30 s / 120 s / 300 s; 3 Brutplätze.
- Brüten: 300 s, 1 Nest; Art 50/50, Mischart 5 % (Lava-Axolotl × Stein-Schildkröte → Glutschildkröte, Glühqualle × Donner-Hamster → Sturmqualle); Gene `clamp(round(avg + rand(-6..10)), 1, 100)`; Seltenheit = max der Eltern, 3 % eine Stufe höher; Mutation 5 % + 5 % je mutiertem Elternteil, bei Mutation 50 % Typ eines mutierten Elternteils erben.
- Ein Wesen ist entweder Begleiter, im Garten oder im Lager. Begleiter dürfen nicht ins Nest; Wesen im Nest dürfen nicht Begleiter werden.
- Alle Wahrscheinlichkeiten (Ei-Seltenheit mit aktuellem Glück, Mutation beim Schlüpfen) sind im Eier-Tab sichtbar.
- Zeiten serverseitig mit `os.time()`; Client rechnet Restzeit mit `workspace:GetServerTimeNow()`.

```lua
local Rarity = require(script.Parent.Config.Rarity)
local Species = require(script.Parent.Config.Species)
local Mutations = require(script.Parent.Config.Mutations)
local Creatures = require(script.Parent.Config.Creatures)

local Genetics = {}

Genetics.GENES = { "luck", "dig", "carry", "income" }

local function mutationMultiplier(mutation: string?): number
	return if mutation then Mutations.byId[mutation].multiplier else 1
end

function Genetics.effective(creature, gene: string): number
	return creature.genes[gene] * Rarity.multiplier[creature.rarity] * mutationMultiplier(creature.mutation)
end

function Genetics.rollMutation(rng: Random): string
	local total = 0
	for _, m in Mutations.list do
		total += m.weight
	end
	local r = rng:NextNumber() * total
	for _, m in Mutations.list do
		r -= m.weight
		if r < 0 then
			return m.id
		end
	end
	return Mutations.list[#Mutations.list].id
end

function Genetics.hatch(egg, rng: Random, id: string)
	local range = Creatures.HATCH.geneRange[egg.zone]
	local genes = {}
	for _, gene in Genetics.GENES do
		genes[gene] = rng:NextInteger(range[1], range[2])
	end
	local mutation = nil
	if rng:NextNumber() < Creatures.HATCH.mutationChance then
		mutation = Genetics.rollMutation(rng)
	end
	return {
		id = id,
		species = Species.base[rng:NextInteger(1, #Species.base)],
		rarity = egg.rarity,
		genes = genes,
		mutation = mutation,
		generation = 1,
	}
end

function Genetics.mutationChance(a, b): number
	local chance = Creatures.BREED.mutationBase
	if a.mutation then
		chance += Creatures.BREED.mutationPerParent
	end
	if b.mutation then
		chance += Creatures.BREED.mutationPerParent
	end
	return chance
end

local function childRarity(a, b, rng: Random): string
	local index = math.max(Rarity.index(a.rarity), Rarity.index(b.rarity))
	if rng:NextNumber() < Creatures.BREED.rarityUpChance then
		index = math.min(index + 1, #Rarity.order)
	end
	return Rarity.order[index]
end

function Genetics.breed(a, b, rng: Random, id: string)
	local B = Creatures.BREED
	local species
	local hybrid = Species.hybridOf(a.species, b.species)
	if hybrid and rng:NextNumber() < B.hybridChance then
		species = hybrid
	else
		species = if rng:NextNumber() < 0.5 then a.species else b.species
	end

	local genes = {}
	for _, gene in Genetics.GENES do
		local avg = (a.genes[gene] + b.genes[gene]) / 2
		genes[gene] = math.clamp(math.round(avg + rng:NextInteger(B.geneDelta[1], B.geneDelta[2])), 1, 100)
	end

	local mutation = nil
	if rng:NextNumber() < Genetics.mutationChance(a, b) then
		local inherited = {}
		if a.mutation then
			table.insert(inherited, a.mutation)
		end
		if b.mutation then
			table.insert(inherited, b.mutation)
		end
		if #inherited > 0 and rng:NextNumber() < B.inheritMutationChance then
			mutation = inherited[rng:NextInteger(1, #inherited)]
		else
			mutation = Genetics.rollMutation(rng)
		end
	end

	return {
		id = id,
		species = species,
		rarity = childRarity(a, b, rng),
		genes = genes,
		mutation = mutation,
		generation = math.max(a.generation, b.generation) + 1,
	}
end

function Genetics.displayName(creature): string
	local name = Species.byId[creature.species].name
	if creature.mutation then
		name = Mutations.byId[creature.mutation].name .. "-" .. name
	end
	return name
end

return Genetics
```

local Genetics = require(script.Parent.Genetics)
local Config = require(script.Parent.Config.Economy)

local Economy = {}

function Economy.incomePerSecond(data): number
	local total = 0
	for _, id in data.garden do
		local creature = data.creatures[id]
		if creature then
			total += Genetics.effective(creature, "income") * Config.INCOME_FACTOR
		end
	end
	return total
end

function Economy.offlineEarnings(data, now: number): number
	if data.lastOnline == 0 then
		return 0
	end
	local elapsed = math.clamp(now - data.lastOnline, 0, Config.OFFLINE_MAX_SECONDS)
	return math.floor(elapsed * Config.OFFLINE_RATE * Economy.incomePerSecond(data))
end

return Economy
